// aop_rust_core/src/aop_value.rs

use num_bigint::BigInt;
use pyo3::prelude::*;
use std::collections::HashMap;

// The internal Rust enum. It is NOT exposed to Python.
#[derive(Debug, Clone, PartialEq)]
pub enum CoeffData {
    Literal(BigInt),
    Power { base: BigInt, exponent: BigInt },
}

// The public-facing Python class that WRAPS CoeffData.
#[pyclass(name = "SymbolicCoefficient")]
#[derive(Debug, Clone, PartialEq)]
pub struct SymbolicCoefficientPy {
    pub(crate) data: CoeffData,
}

#[pyclass]
#[derive(Debug, Clone)]
pub struct AoPValue {
    #[pyo3(get, set)]
    pub coeff: SymbolicCoefficientPy,
    pub poly: HashMap<BigInt, BigInt>,
    #[pyo3(get, set)]
    pub base: u32,
}
