use num_bigint::BigInt;
use num_traits::{One, Zero};
use std::collections::HashMap;

// Helper to calculate n!
fn factorial(n: u32) -> BigInt {
    (1..=n).map(BigInt::from).product()
}

// Calculates the multinomial coefficient: n! / (k1! * k2! * ...)
fn calculate_multinomial_coeff(n: u32, k_vector: &[u32]) -> BigInt {
    let n_fact = factorial(n);
    let k_fact_prod: BigInt = k_vector.iter().map(|&k| factorial(k)).product();
    if k_fact_prod.is_zero() {
        return BigInt::zero();
    }
    n_fact / k_fact_prod
}

// --- START OF CORRECTED ITERATOR LOGIC ---

struct KVectorIterator {
    k_vec: Vec<u32>,
    is_done: bool,
}

impl KVectorIterator {
    fn new(n: u32, m: usize) -> Self {
        if m == 0 {
            return KVectorIterator {
                k_vec: vec![],
                is_done: true,
            };
        }
        let mut k_vec = vec![0; m];
        k_vec[0] = n;
        KVectorIterator {
            k_vec,
            is_done: false,
        }
    }
}

impl Iterator for KVectorIterator {
    type Item = Vec<u32>;

    fn next(&mut self) -> Option<Self::Item> {
        if self.is_done {
            return None;
        }

        let result = self.k_vec.clone();

        // Find the first element from the left that is not zero
        let mut i = 0;
        while i < self.k_vec.len() && self.k_vec[i] == 0 {
            i += 1;
        }

        // If no such element is found, or it's the last element, we are done
        if i >= self.k_vec.len() - 1 {
            self.is_done = true;
            return Some(result);
        }

        // The core logic of the "next composition" algorithm
        let x = self.k_vec[i];
        self.k_vec[i] = 0;
        self.k_vec[0] = x - 1;
        self.k_vec[i + 1] += 1;

        Some(result)
    }
}

// --- END OF CORRECTED ITERATOR LOGIC ---

pub fn expand_multinomial(poly: &HashMap<BigInt, BigInt>, n: u32) -> HashMap<BigInt, BigInt> {
    let terms: Vec<_> = poly.iter().collect();
    let num_terms = terms.len();
    if num_terms == 0 {
        return HashMap::new();
    }
    if num_terms == 1 {
        let (base_exp, base_coeff) = terms[0];
        let new_exp = base_exp * BigInt::from(n);
        let new_coeff = base_coeff.pow(n);
        let mut result = HashMap::new();
        result.insert(new_exp, new_coeff);
        return result;
    }

    let k_vector_iter = KVectorIterator::new(n, num_terms);
    let mut final_poly = HashMap::new();

    for k_vec in k_vector_iter {
        let multinomial_coeff = calculate_multinomial_coeff(n, &k_vec);
        let mut term_coeff: BigInt = One::one();
        let mut term_exp: BigInt = Zero::zero();

        for i in 0..num_terms {
            if k_vec[i] > 0 {
                let (base_exp, base_coeff) = terms[i];
                term_coeff *= base_coeff.pow(k_vec[i]);
                term_exp += base_exp * BigInt::from(k_vec[i]);
            }
        }

        let total_coeff = multinomial_coeff * term_coeff;
        *final_poly.entry(term_exp).or_default() += total_coeff;
    }

    final_poly
}
